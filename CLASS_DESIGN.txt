アクションデータは固定で内部に持つ。
完成後、外部データの読み込み機能を追加する。



enum class EffectType{
    Attack      ,//攻撃
    Heal        ,//回復
};

enum class ParamType{
    Physical    ,//物理
    Magic       ,//魔法
};

enum class TargetSide{
    Enemy       ,//敵
    Friend      ,//味方
    Self        ,//自身
};

enum class TargetSelect{
    LowestHp    ,//Hpが低い
    Random      ,//ランダム
};

enum class TargetRange{
    All         ,//全体
    One         ,//単体
};


class Action{
    EffectType mEffectType;
    ParamType mParamType;
    TargetSide mTargetSide;
    TargetSelect mTargetSelect;
    std::optional<TargetRange> mTargetRange;
};

class ActionList{
    std::vector<Action> mList;
};

enum class ActionId{
    AttackOne,
    AttackDouble,
    MagicAttackOne,
    MagicAttackAll,
    HealOne,
    HealAll,
};

class CharacterAction{
    ActionId mId;
    int mPercent;
};


class Character{
private:
    int mHp;
    int mMaxHp;
    int mAttack;
    int mMagic;
    int mDefence;
    int mSpeed;
    std::vector<CharacterAction> mAction;
public:
    Character( int maxHp, int attack, int magic, int defence, int speed ):
        mHp(maxHp),mMaxHp(maxHp),mAttack(attack),mMagic(magic),mDefence(defence),mSpeed(speed);
};
